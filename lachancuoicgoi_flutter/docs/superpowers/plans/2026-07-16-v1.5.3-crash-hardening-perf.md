/// Tracks how many times each subsystem fell back to degraded mode.
/// When count ≥ 3 in a session, the UI shows a "Chế độ giảm thiểu" banner.
class FallbackTracker {
  final Map<String, int> _counts = {};

  void record(String subsystem) {
    _counts[subsystem] = (_counts[subsystem] ?? 0) + 1;
  }

  int count(String subsystem) => _counts[subsystem] ?? 0;

  bool get hasDegradedSubsystem =>
      _counts.values.any((c) => c >= 3);

  void reset() => _counts.clear();
}

```dart
class CachingAssetLoader implements AssetLoader {
  CachingAssetLoader(this._primary, {AssetLoader? fallback})
      : _fallback = fallback;

  final AssetLoader _primary;
  final AssetLoader? _fallback;
  final Map<String, String> _cache = {};

  @override
  Future<String> loadString(String key) async {
    final cached = _cache[key];
    if (cached != null) return cached;
    try {
      final result = await _primary.loadString(key);
      _cache[key] = result;
      return result;
    } on Object catch (_) {
      if (_fallback != null) {
        final result = await _fallback!.loadString(key);
        _cache[key] = result;
        return result;
      }
      rethrow;
    }
  }

  // ... delegate load() to _primary (binary assets not cached)
}
```

  /// Returns the minimum text delta (chars) before re-analyzing, adapted
  /// to the current speech rate. When STT is slow (low chars/sec), we use
  /// a smaller delta so analysis fires sooner — the user sees alerts
  /// faster on slow devices. When STT is fast, we keep the default to
  /// avoid excessive analysis calls.
  int _adaptiveMinDelta(int baseDelta) {
    // _speechRate is chars/sec. Below 5 cps = slow STT (low-end device,
    // noisy audio). Above 15 cps = fast STT (clear audio, fast talker).
    if (_speechRate > 0 && _speechRate < 5) {

      return math.max(10, baseDelta ~/ 2);
    }
    return baseDelta;
  }
